#include <bits/stdc++.h>

using namespace std;

class Solution {
public:
    int countSubmatrices(vector<vector<int>>& grid, int k);
private:
    vector<vector<int>> buildSumMatrix(vector<vector<int>>& grid);
};

int Solution::countSubmatrices(vector<vector<int>>& grid, int k) {
    // 定义行列
    const int row = grid.size();
    const int col = grid[0].size();
    vector<vector<int>> sumMatrix = buildSumMatrix(grid);

    // 遍历矩阵坐标
    int count = 0;
    for (int i = 0; i < row; i++) {
        for (int j = 0; j < col; j++) {
            if (sumMatrix[i][j] <= k) {
                count++;
            }
        }
    }
    return count;
}

vector<vector<int>> Solution::buildSumMatrix(vector<vector<int>>& grid) {
    // 初始化
    const int row = grid.size();
    const int col = grid[0].size();
    vector<vector<int>> sumMatrix(row, vector<int>(col));

    // 初始化前缀和矩阵
    for (int i = 0; i < row; i++) {
        for (int j = 0; j < col; j++) {
            if (i == 0 && j == 0) {
                sumMatrix[i][j] = grid[i][j];
            } else if (i == 0 && j > 0) {
                sumMatrix[i][j] = sumMatrix[i][j - 1] + grid[i][j];
            } else if (i > 0 && j == 0) {
                sumMatrix[i][j] = sumMatrix[i - 1][j] + grid[i][j];
            } else {
                sumMatrix[i][j] = sumMatrix[i - 1][j] + sumMatrix[i][j - 1] - sumMatrix[i - 1][j - 1] + grid[i][j];
            }
        }
    }
    return sumMatrix;
}


int main() {
    vector<vector<int>> grid = {{7, 6, 3}, {6, 6, 1}};
    int k = 18;
    Solution solution;
    cout << solution.countSubmatrices(grid, k);
}
